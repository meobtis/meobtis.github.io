import { getCollection } from 'astro:content';
import { readerUrl, sourceById } from '../../../lib/catalog';
import { introductions } from '../../../lib/introductions';
import { translationsBySlug } from '../../../lib/translations';

export const docPages = (await getCollection('docs'))
	.filter((d) => d.id.startsWith('translations/') && d.id !== 'translations')
	.sort((a, b) => (a.data.sidebar?.order ?? Infinity) - (b.data.sidebar?.order ?? Infinity));

export const itemsBySlug = await translationsBySlug();
export const intros = await introductions();

Selected Sanskrit-to-English translations from the [meobtis-en](https://github.com/meobtis/meobtis-en) repository,
listed in reading order. Each page displays the English translation and includes a switch to show the four-line
interlinear view (Sanskrit, grammatical analysis, word-by-word gloss, and translation).

<ol class="works not-content">
	{
		docPages.map((doc, idx) => {
			const slug = doc.id.replace(/^translations\//, '');
			const item = itemsBySlug.get(slug);
			const intro = item?.work ? intros.get(item.work) : undefined;
			const source = item?.edition ? sourceById[item.edition.source] : undefined;
			return (
				<li class="work">
					<div class="work-main">
						<div class="work-title">
							<span class="badge">{idx + 1}</span>
							<span class="iast" lang="sa-Latn">
								{doc.data.title}
							</span>
							{item?.work?.titleDeva && (
								<span class="deva" lang="sa-Deva">
									{item.work.titleDeva}
								</span>
							)}
						</div>
						{intro?.data.englishTitle && <div class="english">{intro.data.englishTitle}</div>}
						{intro && <p class="summary">{intro.data.summary}</p>}
						<div class="work-meta">
							{source && <span>{source.label}</span>}
							{item?.edition && <a href={readerUrl(item.edition)}>Sanskrit edition</a>}
							{item?.interlinear && <span>Interlinear available</span>}
						</div>
					</div>
					<a class="read" href={`/${doc.id}/`} aria-label={`Read translation of ${doc.data.title}`}>
						Read
					</a>
				</li>
			);
		})
	}
</ol>
